cmake_minimum_required(VERSION 3.22)
project(mshell LANGUAGES C CXX)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

file(GLOB_RECURSE SOURCES
    "src/*.cpp"
    "src/*.c"
)

add_executable(mshell ${SOURCES})

# build.bat creates build/<mode>/bin/, so put the exe there too. Without
# this, Ninja drops mshell.exe one level up in build/<mode>/ instead - easy
# to miss, and easy to end up repeatedly running a stale exe left in bin/
# from an earlier build while thinking you rebuilt.
set_target_properties(mshell PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
)
